# 条件熵

$$
\begin{align}
H(xy) &= \sum_{i,j} \,p(x_iy_j) \cdot I(x_iy_j) \\
&= \sum_i \,\sum_j \,p(x_iy_j) \cdot [I(y_j|x_i)+I(x_i)] \\
&= \sum_i \,\sum_j \,p(x_iy_j) \cdot I(y_j|x_i) + \sum_i \,\sum_j \,p(x_iy_j) \cdot I(x_i) \\
&= \sum_i \,\sum_j \,p(x_i)p((y_j|x_i) \cdot I((y_j|x_i) + \sum_i \,p(x_i) \cdot I(x_i) \\
&= H(y|x) + H(x)\\
其中 \\
&H(x) = \sum_i \,p(x_i) \cdot I(x_i)\\
&H(y|x) = \sum_i \,\sum_j  p(x_i) \,p(y_j|x_i) \cdot I(y_j|x_i)\\
\\

\end{align}
$$
$$
\begin{align}
H(xy) &= H(x) + H(y|x)\\
&= H(y) + H(x|y)
\end{align}
$$
