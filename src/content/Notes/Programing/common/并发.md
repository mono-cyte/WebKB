#### Win API

1. 相关变量
```cpp
// Windows API
HANDLE event = CreateEvent(
						   NULL,    //[in, optional] LPSECURITY_ATTRIBUTES lpEventAttributes,
						   false,   //[in]           BOOL                  bManualReset,
						   false,   //[in]           BOOL                  bInitialState,
						   "event"  //[in, optional] LPCSTR                lpName
						   );
```

2. 线程函数

```cpp
DWORD WINAPI ThreadProc(void* lpParameter) { // 参数,返回值: 固定规范

    HANDLE event = OpenEvent(EVENT_MODIFY_STATE, FALSE, "event");    // 占用
    
    WaitForSingleObject(event, INFINITE);                            // 等待
    
	//...    处理逻辑
	
    setEvent(event);                                                 // 解锁+唤起
    
    return 0;

}
```

3. 线程调度
```cpp

HANDLE thread = CreateThread(
                             NULL,          //[in, optional]  LPSECURITY_ATTRIBUTES   lpThreadAttributes,
                             0,             //[in]            SIZE_T                  dwStackSize,
                             ThreadProc,    //[in]            LPTHREAD_START_ROUTINE  lpStartAddress,
                             NULL,          //[in, optional]  __drv_aliasesMem LPVOID lpParameter,
                             0,             //[in]            DWORD                   dwCreationFlags,
                             NULL           //[out, optional] LPDWORD                 lpThreadId
                            );

setEvent(event);

WaitForSingleObject(thread, INFINITE);
```

##### 忙等待 lock  
1. 相关变量
```cpp
// C++ 标准库
std::mutex mtx;
```
2. 线程函数
```c
void proc(){
	while(1){
		std::unique_lock<std::mutex> lock(mtx);       // 占用+等待, 作用域结束自动解锁(部分类型锁可手动 unlock() )
		//...    处理逻辑
	}
}
```
3. 线程调度

```cpp
std::thread t(proc);
t.join();    // 阻塞线程, 等待
```

##### 通知 condition_variable

1. 相关变量
```cpp
// C++ 标准库
std::mutex mtx;
std::condition_variable cv;
bool in_use = false; // 占用状态or信号
```
2. 线程函数
```cpp
void proc(){
	while(1){
		std::unique_lock<std::mutex> lock(mtx);       // 占用+等待
		cv.wait(lock, []{ return in_use; });          // 等待 (被通知前保持线程阻塞)
		//...    处理逻辑
		in_use = false;                               // 解锁
		cv.notify_all();                              // 唤起
	}
}
```
3. 线程调度

```cpp
std::thread t(proc);
t.join();    // 阻塞线程, 等待
```

##### 信号量 std::semaphore

> 要求: c++20
```cpp
std::counting_semaphore<> workstations(3); // 3线程

void threadProc(int id) {
    workstations.acquire(); // 等待一个空闲工位
	//...    处理逻辑
    workstations.release(); // 释放工位
}

int main() {
    std::vector<std::jthread> threads; // jthread 析构时自动 join
    for (int i = 0; i < 10; ++i) {
        threads.emplace_back(threadProc, i);
    }
    
    return 0;
}
```

##### std::future & std:promise

1. 相关变量
```cpp
std::promise<int> prom;
std::future<int> fut = prom.get_future();
```

2. 线程函数

```cpp
void proc(std::promise<int> promise){

	// ...   处理逻辑
	
	promise.set_value(0);
}
```


3. 线程调度
```cpp
std::thread t(proc, std::move(prom));    // promise 不可复制,可移动: 移动所有权,防止复杂情况下指针悬空
int result = fut.get();                  // 阻塞线程, 等待
t.join();                                // 阻塞线程, 等待
```
