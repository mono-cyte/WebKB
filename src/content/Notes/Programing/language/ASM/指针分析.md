# 指针分析
| asm                               | c                              |
| --------------------------------- | ------------------------------ |
| `mov dword ptr [ebp - 0x8], 0xf`  | 局部变量1 `int a = 0xF`            |
| `lea eax, [ebp - 0x8]`            | `&a`                           |
| `mov dword ptr [ebp - 0xc], eax`  | 局部变量2 `int* p = &a`            |
| `lea eax, [ebp - 0xc]`            | `&p`                           |
| `mov dword ptr [ebp - 0x10], eax` | 局部变量3 `int** pp = &p`          |
| `lea eax, [ebp - 0x10]`           | `&pp`                          |
| `mov dword ptr [ebp - 0x14], eax` | 局部变量4 `int*** ppp = &pp`       |
| `lea eax, [ebp - 0x14]`           | `&ppp`                         |
| `mov dword ptr [ebp - 0x18], eax` | 局部变量5 `int**** pppp = &ppp`    |
| `lea eax, [ebp - 0x18]`           | `&pppp`                        |
| `mov dword ptr [ebp - 0x1c], eax` | 局部变量6 `int***** ppppp = &pppp` |
| `mov eax, dword ptr [ebp - 0x1c]` | `ppppp`                        |
| `mov eax, dword ptr [eax]`        | `*ppppp`                       |
| `mov dword ptr [ebp - 0x20], eax` | 局部变量7 `int**** pppp1 = *ppppp` |
| `mov eax, dword ptr [ebp - 0x20]` | `pppp1`                        |
| `mov eax, dword ptr [eax]`        | `*pppp1`                       |
| `mov dword ptr [ebp - 0x24], eax` | 局部变量8 `int*** ppp1 = *pppp1`   |
| `mov eax, dword ptr [ebp - 0x24]` | `ppp1`                         |
| `mov eax, dword ptr [eax]`        | `*ppp1`                        |
| `mov dword ptr [ebp - 0x28], eax` | 局部变量9 `int** pp1 = *ppp1`      |
| `mov eax, dword ptr [ebp - 0x28]` | `pp1`                          |
| `mov eax, dword ptr [eax]`        | `*pp1`                         |
| `mov dword ptr [ebp - 0x2c], eax` | 局部变量10 `int* p1 = *pp1`        |
| `mov eax, dword ptr [ebp - 0x2c]` | `p1`                           |
| `mov eax, dword ptr [eax]`        | `*p1`                          |
| `mov dword ptr [ebp - 0x30], eax` | 局部变量11 `int a1 = *p1`          |

```c
    int a = 0xF;
    int* p = &a;
    int** pp = &p;
    int*** ppp = &pp;
    int**** pppp = &ppp;
    int***** ppppp = &pppp;

    int**** pppp1 = *ppppp;
    int*** ppp1 = *pppp1;
    int** pp1 = *ppp1;
    int* p1 = *pp1;
    int a1 = *p1;
```
